starbucks_green = '#00704A'

portfolio = pd.DataFrame({
    'Company': ['Starbucks', "Dunkin' Donuts"],
    'Beverage_combinations': [len(sbux4), len(dnkn4)],
})

ax = portfolio.plot(
    kind='barh',
    x='Company',
    y='Beverage_combinations',
    color=[starbucks_green, dunkin_pink],
    figsize=(8, 5),
    legend=False,
)
plt.title('Beverage Combination Portfolio Size', fontsize=14, fontweight='bold')
plt.xlabel('Number of Beverage Combinations')

for p in ax.patches:
  ax.annotate(
      str(int(p.get_width())),
      (p.get_width() / 2, p.get_y() + p.get_height() / 2),
      ha='center',
      va='center',
      color='white',
      fontweight='bold',
  )
plt.show()

avg_calories = pd.DataFrame({
